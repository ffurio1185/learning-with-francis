SELECT name, city FROM learners;

{{ _('This returns Alex/Berlin, Mia/Hamburg and Sam/Berlin. It reads rows without changing them.') }}

SELECT * FROM learners;

{{ _('The star') }} * {{ _('means every column: name, city and age.') }}

{{ _('2. Choose rows with WHERE') }}

WHERE {{ _('keeps only rows that match a condition. Text values use single quotes; numbers do not need quotes.') }}

SELECT name FROM learners WHERE city = 'Berlin';

{{ _('This returns Alex and Sam.') }} = {{ _('means equal to.') }}

SELECT name FROM learners WHERE age >= 18;

{{ _('This also returns Alex and Sam.') }} >= {{ _('means greater than or equal to;') }} < {{ _('means less than;') }} > {{ _('means greater than.') }}

SELECT name FROM learners WHERE age < 18;
